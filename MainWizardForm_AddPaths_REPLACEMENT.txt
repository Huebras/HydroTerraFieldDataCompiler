Replace the existing AddPaths(IEnumerable<string> paths) method in MainWizardForm.cs with this method:

    private void AddPaths(IEnumerable<string> paths)
    {
        SurveyImportResolution resolution = SurveyImportResolver.Resolve(paths);

        foreach (string logPath in resolution.LogPaths)
        {
            if (!_project.ImportedLogFiles.Contains(logPath, StringComparer.OrdinalIgnoreCase))
                _project.ImportedLogFiles.Add(logPath);
        }

        foreach (string surveyPath in resolution.RawOrZipPaths)
        {
            if (!_project.ImportedRawFiles.Contains(surveyPath, StringComparer.OrdinalIgnoreCase))
                _project.ImportedRawFiles.Add(surveyPath);
        }

        if (resolution.RawFilesDiscoveredFromLogs.Count > 0)
        {
            _project.Findings.RemoveAll(f => f.RuleId == "LOG-RAW-AUTOLOAD");
            _project.Findings.Add(new QaFinding
            {
                RuleId = "LOG-RAW-AUTOLOAD",
                Severity = "Info",
                Category = "HYPACK LOG",
                Description = $"Loaded {resolution.RawFilesDiscoveredFromLogs.Count} RAW file(s) referenced by the selected HYPACK LOG file(s).",
                Evidence = string.Join(", ", resolution.RawFilesDiscoveredFromLogs.Select(Path.GetFileName).Take(30))
            });
        }

        foreach (string unresolved in resolution.UnresolvedLogReferences)
        {
            if (_project.Findings.Any(f => f.RuleId == "LOG-RAW-UNRESOLVED" && f.Evidence == unresolved))
                continue;

            _project.Findings.Add(new QaFinding
            {
                RuleId = "LOG-RAW-UNRESOLVED",
                Severity = "Warning",
                Category = "HYPACK LOG",
                Description = "A RAW file referenced by the selected HYPACK LOG could not be found beside the LOG or below its survey folder.",
                Evidence = unresolved
            });
        }
    }
